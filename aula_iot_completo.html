from django.db import models
from django.utils import timezone


class Aluno(models.Model):
    """Representa um aluno cadastrado no sistema."""
    nome       = models.CharField(max_length=100)
    matricula  = models.CharField(max_length=20, unique=True)
    uid_rfid   = models.CharField(max_length=50, unique=True,
                    help_text="UID da tag/cartão RFID")
    turma      = models.CharField(max_length=20)
    ativo      = models.BooleanField(default=True)
    criado_em  = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return f"{self.nome} ({self.matricula})"

    class Meta:
        ordering = ['nome']
        verbose_name_plural = 'Alunos'


class Livro(models.Model):
    """Representa um livro do acervo."""
    titulo     = models.CharField(max_length=200)
    autor      = models.CharField(max_length=150)
    isbn       = models.CharField(max_length=20, blank=True)
    uid_rfid   = models.CharField(max_length=50, unique=True,
                    help_text="UID da tag RFID colada no livro")
    disponivel = models.BooleanField(default=True)
    criado_em  = models.DateTimeField(auto_now_add=True)

    def __str__(self):
        return f"{self.titulo} — {self.autor}"

    class Meta:
        ordering = ['titulo']


class Emprestimo(models.Model):
    """Registro de empréstimo criado automaticamente via RFID."""
    aluno          = models.ForeignKey(Aluno, on_delete=models.CASCADE,
                        related_name='emprestimos')
    livro          = models.ForeignKey(Livro, on_delete=models.CASCADE,
                        related_name='emprestimos')
    data_retirada  = models.DateTimeField(auto_now_add=True)
    data_devolucao = models.DateTimeField(null=True, blank=True)
    prazo          = models.DateField(null=True, blank=True)
    ativo          = models.BooleanField(default=True)

    @property
    def atrasado(self):
        from datetime import date
        if self.ativo and self.prazo:
            return date.today() > self.prazo
        return False

    def __str__(self):
        return f"{self.aluno.nome} → {self.livro.titulo}"

    class Meta:
        ordering = ['-data_retirada']


class LogAcesso(models.Model):
    """Log de leituras RFID (debugging e auditoria)."""
    uid_lido   = models.CharField(max_length=50)
    aluno      = models.ForeignKey(Aluno, on_delete=models.SET_NULL,
                    null=True, blank=True)
    livro      = models.ForeignKey(Livro, on_delete=models.SET_NULL,
                    null=True, blank=True)
    timestamp  = models.DateTimeField(auto_now_add=True)
    status     = models.CharField(max_length=20,
                    choices=[(
                        'sucesso', 'Sucesso'),(
                        'erro', 'Erro'),(
                        'nao_encontrado', 'Não Encontrado')])
    mensagem   = models.TextField(blank=True)

    def __str__(self):
        return f"[{self.timestamp}] {self.uid_lido} — {self.status}"

biblioteca/views.py
from django.http import JsonResponse
from django.views.decorators.csrf import csrf_exempt
from django.shortcuts import render
from datetime import date, timedelta
import json
from .models import Aluno, Livro, Emprestimo, LogAcesso


# ──────────────────────────────────────────────
#  ENDPOINT: Registrar Empréstimo via RFID
#  Chamado pelo ESP32 quando aluno + livro são lidos
# ──────────────────────────────────────────────
@csrf_exempt
def api_emprestimo(request):
    if request.method != 'POST':
        return JsonResponse({'erro': 'Método inválido'}, status=405)

    try:
        dados = json.loads(request.body)
    except json.JSONDecodeError:
        return JsonResponse({'erro': 'JSON inválido'}, status=400)

    uid_aluno = dados.get('uid_aluno')
    uid_livro = dados.get('uid_livro')

    # Registrar no log independente do resultado
    log = LogAcesso(uid_lido=f"{uid_aluno}+{uid_livro}")

    try:
        aluno = Aluno.objects.get(uid_rfid=uid_aluno, ativo=True)
        livro = Livro.objects.get(uid_rfid=uid_livro, disponivel=True)
    except Aluno.DoesNotExist:
        log.status = 'nao_encontrado'; log.mensagem = 'Aluno não encontrado'
        log.save()
        return JsonResponse({'erro': 'Aluno não cadastrado'}, status=404)
    except Livro.DoesNotExist:
        log.status = 'nao_encontrado'; log.mensagem = 'Livro indisponível'
        log.save()
        return JsonResponse({'erro': 'Livro não disponível'}, status=404)

    # Criar empréstimo com prazo de 7 dias
    emp = Emprestimo.objects.create(
        aluno=aluno,
        livro=livro,
        prazo=date.today() + timedelta(days=7)
    )
    livro.disponivel = False
    livro.save()

    log.aluno = aluno; log.livro = livro
    log.status = 'sucesso'
    log.save()

    return JsonResponse({
        'mensagem': 'Empréstimo registrado com sucesso!',
        'aluno':    aluno.nome,
        'livro':    livro.titulo,
        'prazo':    str(emp.prazo),
        'id':       emp.pk
    }, status=201)


# ──────────────────────────────────────────────
#  ENDPOINT: Devolver livro via RFID
# ──────────────────────────────────────────────
@csrf_exempt
def api_devolucao(request):
    if request.method != 'POST':
        return JsonResponse({'erro': 'Método inválido'}, status=405)

    dados = json.loads(request.body)
    uid_livro = dados.get('uid_livro')

    try:
        livro = Livro.objects.get(uid_rfid=uid_livro)
        emp = Emprestimo.objects.get(livro=livro, ativo=True)
    except (Livro.DoesNotExist, Emprestimo.DoesNotExist):
        return JsonResponse({'erro': 'Empréstimo não encontrado'}, status=404)

    from django.utils import timezone
    emp.data_devolucao = timezone.now()
    emp.ativo = False
    emp.save()
    livro.disponivel = True
    livro.save()

    return JsonResponse({
        'mensagem': 'Devolução registrada!',
        'livro': livro.titulo,
        'aluno': emp.aluno.nome
    })


# ──────────────────────────────────────────────
#  VIEW: Dashboard HTML
# ──────────────────────────────────────────────
def dashboard(request):
    context = {
        'total_livros':       Livro.objects.count(),
        'livros_disponiveis': Livro.objects.filter(disponivel=True).count(),
        'total_alunos':       Aluno.objects.filter(ativo=True).count(),
        'emprestimos_ativos': Emprestimo.objects.filter(ativo=True).count(),
        'emprestimos':        Emprestimo.objects.filter(ativo=True).select_related(),
        'ultimos_logs':       LogAcesso.objects.order_by('-timestamp')[:10],
    }
    return render(request, 'biblioteca/dashboard.html', context)

esp32/main.py — código completo
esp32/main.py
# ─────────────────────────────────────────────
#  ESP32 — Sistema de Empréstimo RFID
#  Linguagem: MicroPython
# ─────────────────────────────────────────────
import network
import urequests
import ujson
import time
import machine
from mfrc522 import MFRC522
from config import *


# ─── Configuração dos pinos ───
led_ok   = machine.Pin(PIN_LED_OK,   machine.Pin.OUT)
led_erro = machine.Pin(PIN_LED_ERRO, machine.Pin.OUT)
buzzer   = machine.Pin(PIN_BUZZER,   machine.Pin.OUT)


# ─── Conectar ao WiFi ───
def conectar_wifi():
    wlan = network.WLAN(network.STA_IF)
    wlan.active(True)
    if not wlan.isconnected():
        print("Conectando ao WiFi...")
        wlan.connect(WIFI_SSID, WIFI_SENHA)
        while not wlan.isconnected():
            time.sleep(0.5)
            print(".", end="")
    print("\nWiFi conectado! IP:", wlan.ifconfig()[0])
    return wlan


# ─── Ler UID do RFID ───
def ler_uid(rfid):
    rfid.init()
    (stat, tag_type) = rfid.request(rfid.REQIDL)
    if stat == rfid.OK:
        (stat, uid) = rfid.SelectTagSN()
        if stat == rfid.OK:
            return "".join(["{:02X}".format(x) for x in uid])
    return None


# ─── Feedbacks ───
def bip_sucesso():
    led_ok.on(); buzzer.on()
    time.sleep_ms(200)
    buzzer.off(); time.sleep_ms(100)
    buzzer.on(); time.sleep_ms(200)
    buzzer.off(); time.sleep_ms(800)
    led_ok.off()

def bip_erro():
    led_erro.on(); buzzer.on()
    time.sleep_ms(600)
    buzzer.off(); time.sleep_ms(600)
    led_erro.off()


# ─── Enviar para a API ───
def registrar_emprestimo(uid_aluno, uid_livro):
    payload = {"uid_aluno": uid_aluno, "uid_livro": uid_livro}
    try:
        resp = urequests.post(
            API_EMPRESTIMO,
            data=ujson.dumps(payload),
            headers={"Content-Type": "application/json"},
            timeout=5
        )
        dados = resp.json()
        resp.close()

        if resp.status_code == 201:
            print("✓ Empréstimo:", dados['aluno'], "→", dados['livro'])
            bip_sucesso()
        else:
            print("✗ Erro:", dados.get('erro'))
            bip_erro()

    except Exception as e:
        print("Falha na requisição:", e)
        bip_erro()


# ─── MAIN LOOP ───
def main():
    conectar_wifi()

    rfid = MFRC522(
        sck=PIN_SCK, mosi=PIN_MOSI,
        miso=PIN_MISO, rst=PIN_RST, cs=PIN_SDA
    )
    print("Sistema pronto! Aproxime o cartão do aluno...")

    while True:
        # 1. Leitura do aluno
        uid_aluno = None
        while not uid_aluno:
            uid_aluno = ler_uid(rfid)
            time.sleep_ms(DELAY_LEITURA)

        print("Aluno lido:", uid_aluno)
        print("Agora aproxime o livro...")
        time.sleep(1)

        # 2. Leitura do livro
        uid_livro = None
        tentativas = 0
        while not uid_livro and tentativas < 10:
            uid_livro = ler_uid(rfid)
            time.sleep_ms(DELAY_LEITURA)
            tentativas += 1

        if uid_livro:
            registrar_emprestimo(uid_aluno, uid_livro)
        else:
            print("Timeout — livro não lido")
            bip_erro()

main()

Atividade Prática — Divisão em Grupos

Roteiro de aula com checklists por grupo e critérios de avaliação

Divisão da turma — 4 grupos
🗄️
Grupo 1 — Backend
Django · Models · Admin
Banco de dados · Migrações
🔗
Grupo 2 — API REST
Views · URLs · JSON
Endpoints · Testes Postman
🧠
Grupo 3 — IoT
ESP32 · RFID · Circuito
MicroPython · WiFi
🖥️
Grupo 4 — Frontend
Templates Django · Bootstrap
Dashboard · Gráficos
✅ Checklist — Grupo 1 (Backend)
  • Clonar o repositório do GitHub
  • Criar e ativar o ambiente virtual
  • Instalar dependências (pip install -r requirements.txt)
  • Criar projeto Django (startproject)
  • Criar app biblioteca (startapp)
  • Definir models: Aluno, Livro, Emprestimo, LogAcesso
  • Registrar models no admin.py
  • Rodar makemigrations e migrate
  • Criar superusuário
  • Popular banco com dados de teste via admin
  • Fazer commit e push para o GitHub
✅ Checklist — Grupo 2 (API)
  • Criar views.py com api_emprestimo
  • Criar views.py com api_devolucao
  • Criar biblioteca/urls.py com as rotas
  • Incluir URLs no urls.py principal
  • Testar POST /api/emprestimo/ no Postman
  • Testar resposta de sucesso (201)
  • Testar resposta de erro (404)
  • Verificar log criado no banco
  • Documentar os endpoints no README
  • Fazer commit e push para o GitHub
✅ Checklist — Grupo 3 (IoT)
  • Montar o circuito ESP32 + RC522
  • Verificar pinagem (SDA, SCK, MOSI, MISO, RST)
  • Instalar MicroPython no ESP32
  • Criar esp32/config.py com WiFi e URL da API
  • Criar esp32/main.py com o loop principal
  • Testar leitura de UID no terminal serial
  • Conectar ESP32 ao WiFi
  • Fazer POST para a API e verificar resposta
  • Testar feedback LED verde (sucesso) e vermelho (erro)
  • Testar buzzer
  • Fazer commit do código no GitHub
✅ Checklist — Grupo 4 (Frontend)
  • Criar base.html com navbar e Bootstrap
  • Criar dashboard.html estendendo base
  • Exibir cards com estatísticas (total livros, disponíveis, etc.)
  • Criar tabela de empréstimos ativos
  • Destacar em vermelho os empréstimos atrasados
  • Configurar TEMPLATES no settings.py
  • Criar view dashboard(request) passando contexto
  • Adicionar rota no urls.py
  • Exibir últimos logs de acesso RFID
  • Responsividade mobile
  • Fazer commit e push para o GitHub
🎯 Critérios de avaliação

✅ O que avalia

  • Sistema funcionando de ponta a ponta
  • Leitura RFID registra no banco
  • Dashboard mostra dados reais
  • Código no GitHub com histórico de commits
  • README com instruções de instalação
  • Apresentação explicando a arquitetura

❌ Erros comuns

  • IoT sem integração com Django
  • Complicar demais (MQTT, cloud, etc.)
  • Grupos isolados sem comunicar interfaces
  • Não testar a API antes de ligar o hardware
  • Hardcodar IPs sem usar config.py
  • Esquecer o .gitignore e enviar o venv/
🏁

Sequência recomendada de integração

1. Grupo 1 popula o banco com alunos e livros → 2. Grupo 2 testa API com Postman usando UIDs reais → 3. Grupo 3 conecta o ESP32 à API já testada → 4. Grupo 4 abre o dashboard e mostra o empréstimo registrado em tempo real.
Resultado: toda a turma vê o sistema funcionando do hardware ao navegador. 🎉
